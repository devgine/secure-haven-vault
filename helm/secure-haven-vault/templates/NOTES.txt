  kubectl -n {{ .Release.Namespace }} get cluster {{ include "secure-haven-vault.postgresqlName" . }}

  kubectl -n {{ .Release.Namespace }} logs deploy/{{ include "secure-haven-vault.fullname" . }} | grep migrations_applied

« {{ .Values.config.admin.passwordKey }} » du secret applicatif.
